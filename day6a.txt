1) Create a procedure P1 that inserts numbers from 1 to 10 into table t1 which has only  one column named a of int type. Use while loop only.

Create table t1 (a int);
DELIMITER //

Create procedure p2 ()
BEGIN
	DECLARE counter INT DEFAULT 1;


WHILE counter <= 10
DO

	Insert into t1 values (counter);
	SET counter = counter+1;
END WHILE;

END //
DELIMITER ;

create table t3(c int);

DELIMITER //
CREATE PROCEDURE P3()
BEGIN 

DECLARE COUNT INT DEFAULT 1;
DECLARE SUM INT DEFAULT 0;

WHILE COUNT < 100
DO

SET SUM = SUM + COUNT;
SET COUNT  = COUNT +1;
END WHILE;

7) Create a procedure P7 that inserts squares of numbers from 1 to 15 into a table t6  which has only one column named g of int type. Use basic loop only.




DELIMITER //
CREATE PROCEDURE P7()
BEGIN

DECLARE CON INT DEFAULT 1;
DECLARE NUM INT DEFAULT 1;

BASIC_LOOP: LOOP
	SET NUM = CON ; 
	SET NUM = NUM*NUM ;
	INSERT INTO T7 values(NUM);
	SET CON = CON+1;

	END LOOP BASIC_LOOP;
